int yylex();

%}

%%

    {
        printf("Valid do-while statement\n");
    }
    ;

STMT : ID '=' EXPR ';'
     ;

EXPR : ID OP NUM
     | ID OP ID
     | NUM
     ;

     | ID OP ID
     ;

%%

    yyparse();

}
